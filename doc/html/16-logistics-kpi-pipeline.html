    TRADER["Trader Dashboard<br/>own shipments only"]:::trader
    DRIVER["Driver Dashboard<br/>own operational metrics"]:::driver
  end

  STALE["Visible stale / error state<br/>Never silently freeze old data"]:::warn

  SHIP & EVT & INC & ALT & DEV & GPS --> KPI
  KPI --> TZ
  KPI --> FX
  TZ --> K1 & K2 & K3 & K4
  FX --> K5
  K1 & K2 & K3 & K4 & K5 --> SCOPE
  KPI --> FRESH --> SCOPE
  SCOPE --> ADMIN & TRADER & DRIVER
  FRESH -. "poll/realtime failure" .-> STALE
  STALE -.-> ADMIN & TRADER & DRIVER
